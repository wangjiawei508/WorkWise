"""Emit a generic audit partial reference from the oracle alone, not product output."""
import argparse
import json
from pathlib import Path

parser = argparse.ArgumentParser()
parser.add_argument("oracle", type=Path)
parser.add_argument("output", type=Path)
args = parser.parse_args()
oracle = json.loads(args.oracle.read_text())
station = oracle["solutions"]["rss"]["station"]
reference = {
    "observationCount": 5,
    "unknownCount": 3,
    "degreesOfFreedom": 2,
    "redundancy": 2,
    "linearUnit": "m",
    "angularUnit": "rad",
    "solverDiagnostics": {"rank": 3},
    "points": [{"id": name, "x": coordinates[0], "y": coordinates[1]} for name, coordinates in oracle["fixedControls"].items()] + [{"id": station["id"], "x": station["xNorthM"], "y": station["yEastM"]}],
}
args.output.write_text(json.dumps(reference, indent=2) + "\n")
print(json.dumps({"referenceResultPath": str(args.output), "numericTolerance": oracle["tolerancesDeclaredBeforeProductComparison"]["stationCoordinateAbsoluteM"], "scope": "counts, rank, units, fixed points and adjusted station coordinates only; run compare_result.py for separate angular/linear residual tolerances"}, indent=2))
