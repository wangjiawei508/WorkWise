import { readFile, realpath, stat, lstat, mkdtemp, writeFile, rm } from 'node:fs/promises';
import { join, relative, resolve, sep, isAbsolute } from 'node:path';
import { DatabaseSync } from 'node:sqlite';

// Exactly the five tools advertised for this reviewed synthetic typed plan.
const NAMES = ['survey_read_context', 'survey_network_validate', 'control_network', 'survey_adjustment_read', 'report_export'];
const MARKER = 'RAILWISE-SYNTHETIC-ACCEPTANCE';
const LIMIT = 8 * 1024 * 1024;
const SYNTHETIC_SOURCE_SHA256 = '4281cc7a10673570756867e8c8665f22f82611839bd34c396a5eba9dbb89d9f0';
const deny = code => Object.assign(new Error(code), { publicStatus: 400, publicCode: code });
const canonical = value => JSON.stringify(value && typeof value === 'object' ? Array.isArray(value) ? value.map(v => JSON.parse(canonical(v))) : Object.fromEntries(Object.keys(value).sort().map(key => [key, JSON.parse(canonical(value[key]))])) : value);
const wireNames = new Map();
for (const name of NAMES) {
  wireNames.set(name, name);
  const alias = `railwise.${name}`;
  let hash = 0x811c9dc5;
  for (let i = 0; i < alias.length; i++) { hash ^= alias.charCodeAt(i); hash = Math.imul(hash, 0x01000193); }
  wireNames.set(alias, name);
  wireNames.set(`ww_railwise_${name}_${(hash >>> 0).toString(16).padStart(8, '0')}`, name);
}
const toolName = name => { if (!wireNames.has(name)) throw deny('tool_not_allowed'); return wireNames.get(name); };
const parse = text => { try { return JSON.parse(text); } catch { throw deny('invalid_json'); } };
function contentText(content, nullable = false) {
  if (content == null && nullable) return '';
  if (typeof content === 'string') return content;
  if (Array.isArray(content) && content.every(item => item?.type === 'text' && typeof item.text === 'string')) return content.map(item => item.text).join('\n');
  throw deny('text_only_acceptance');
}
async function scopedPath(path, root) {
  const resolved = await realpath(path);
  const child = relative(root, resolved);
  if (!child || child === '..' || child.startsWith(`..${sep}`) || isAbsolute(child)) throw deny('candidate_scope_invalid');
  return resolved;
}
async function scopedFile(path, root) {
  const resolved = await scopedPath(path, root);
  const info = await stat(resolved);
  if (!info.isFile() || info.size > LIMIT) throw deny('candidate_evidence_invalid');
  return readFile(resolved, 'utf8');
}
export async function scopedDatabase(path, root, query, id, { afterCapture } = {}) {
  const resolved = await scopedPath(path, root);
  if (!(await stat(resolved)).isFile()) throw deny('candidate_evidence_invalid');
  const capture = async () => {
    const snapshot = new Map();
    for (const suffix of ['', '-wal', '-shm', '-journal']) {
      try {
        const info = await lstat(resolved + suffix);
        const scoped = await scopedPath(resolved + suffix, root);
        if (!info.isFile() || info.isSymbolicLink() || info.size > 64 * 1024 * 1024) throw deny('candidate_evidence_invalid');
        if (suffix === '-journal' && info.size) throw deny('candidate_evidence_busy');
        if (suffix === '' || suffix === '-wal') snapshot.set(suffix, await readFile(scoped));
      } catch (error) {
        if (error.code !== 'ENOENT' || !suffix) throw error;
        snapshot.set(suffix, null);
      }
    }
    return snapshot;
  };
  const equal = (left, right) => ['', '-wal'].every(key => left.get(key) === null ? right.get(key) === null : right.get(key) !== null && left.get(key).equals(right.get(key)));
  // SQLite readOnly may create source WAL/SHM files. Open only independently copied snapshots.
  for (let attempt = 0; attempt < 3; attempt++) {
    const captured = await capture();
    await afterCapture?.({ databasePath: resolved, attempt });
    if (!equal(captured, await capture())) continue;
    const temporary = await mkdtemp('/private/tmp/railwise-relay-db-');
    let db;
    try {
      for (const suffix of ['', '-wal']) if (captured.get(suffix) !== null) await writeFile(join(temporary, `snapshot.sqlite3${suffix}`), captured.get(suffix), { flag: 'wx', mode: 0o600 });
      db = new DatabaseSync(join(temporary, 'snapshot.sqlite3'), { readOnly: true });
      const row = db.prepare(query).get(id);
      if (!equal(captured, await capture())) continue;
      if (!row || typeof row.data_json !== 'string' || Buffer.byteLength(row.data_json) > LIMIT) throw deny('candidate_record_missing');
      return { row, value: parse(row.data_json) };
    } finally { db?.close(); await rm(temporary, { recursive: true, force: true }); }
  }
  throw deny('candidate_evidence_busy');
}
export function candidateEvidenceLoader({ candidateRoot, projectId, networkId, threadId }) {
  return async () => {
    if (!/^\/private\/tmp\/railwise-survey-[a-zA-Z0-9-]+$/.test(candidateRoot ?? '') || await realpath(candidateRoot) !== resolve(candidateRoot) || ![projectId, networkId, threadId].every(id => /^[a-zA-Z0-9_-]{1,160}$/.test(id ?? ''))) throw deny('candidate_scope_invalid');
    const runtime = join(candidateRoot, 'home/.workwise/runtime');
    const thread = join(runtime, 'threads', threadId);
    const entries = (await scopedFile(join(thread, 'metadata.jsonl'), candidateRoot)).split(/\r?\n/).filter(Boolean).map(parse);
    const metadata = entries.findLast(entry => entry.kind === 'thread_metadata')?.thread;
    if (metadata?.id !== threadId || metadata.domain !== 'engineering' || metadata.projectId !== projectId || typeof metadata.workspace !== 'string') throw deny('candidate_thread_mismatch');
    const workspace = await scopedPath(metadata.workspace, candidateRoot);
    if (!(await stat(workspace)).isDirectory()) throw deny('candidate_scope_invalid');
    const { row: projectRow, value: project } = await scopedDatabase(join(runtime, 'engineering/engineering.sqlite3'), candidateRoot, 'SELECT id, revision, data_json FROM engineering_projects WHERE id = ?', projectId);
    const { row: networkRow, value: network } = await scopedDatabase(join(runtime, 'engineering/survey.sqlite3'), candidateRoot, 'SELECT id, project_id, revision, data_json FROM survey_networks WHERE id = ?', networkId);
    if (project.id !== projectId || projectRow.id !== projectId || projectRow.revision !== project.revision || !/^审批卡.*（合成数据）$/.test(project.name)
      || typeof project.workspace !== 'string' || await scopedPath(project.workspace, candidateRoot) !== workspace
      || network.id !== networkId || networkRow.id !== networkId || network.projectId !== projectId || networkRow.project_id !== projectId || networkRow.revision !== network.revision || network.networkType !== 'plane-control'
      || network.sourceFile?.name !== 'golden-plane-control-e2e.in2' || network.sourceFile?.sha256 !== SYNTHETIC_SOURCE_SHA256) throw deny('synthetic_fixture_required');
    let messageText = '';
    const messagesPath = join(thread, 'messages.jsonl');
    // A newly created real thread has metadata/events but no message file yet.
    try { await lstat(messagesPath); messageText = await scopedFile(messagesPath, candidateRoot); }
    catch (error) {
      if (error.code !== 'ENOENT') throw error;
      try { await lstat(messagesPath); throw deny('candidate_scope_invalid'); }
      catch (missing) { if (missing.code !== 'ENOENT') throw missing; }
    }
    const messages = messageText.split(/\r?\n/).filter(Boolean).map(parse);
    return { metadata, messages };
  };
}

export function createGuard({ projectId, networkId, threadId, candidateRoot, evidenceLoader = candidateEvidenceLoader({ projectId, networkId, threadId, candidateRoot }) }) {
  if (![projectId, networkId, threadId].every(id => /^[a-zA-Z0-9_-]{1,160}$/.test(id ?? ''))) throw deny('explicit_scope_required');
  const issued = new Map();
  const adjustments = new Set();
  function scope(value) {
    if (typeof value === 'string') {
      if (/(?:https?|ftp|file):\/\/|data:[^\s]{0,80};base64/i.test(value)) throw deny('url_or_attachment_not_allowed');
      return;
    }
    if (!value || typeof value !== 'object') return;
    if (Array.isArray(value)) { value.forEach(scope); return; }
    for (const [key, child] of Object.entries(value)) {
      if (key === 'projectId' && child !== projectId || key === 'networkId' && child !== networkId) throw deny('cross_project_or_network');
      if (/attachment|image_url|audio_url|file_url/i.test(key) && !(key === 'inputAttachmentHash' && child === SYNTHETIC_SOURCE_SHA256)) throw deny('attachment_not_allowed');
      scope(child);
    }
  }
  function callArguments(call) {
    if (!call || call.type !== 'function' || typeof call.id !== 'string' || !call.id || typeof call.function?.arguments !== 'string') throw deny('invalid_tool_call');
    const name = toolName(call.function.name);
    const args = parse(call.function.arguments);
    if (!args || typeof args !== 'object' || Array.isArray(args)) throw deny('invalid_tool_arguments');
    const selectionFields = ['networkId', 'networkRevision', 'sourceSha256', 'adjustmentId', 'observationId', 'sourceRecordId', 'pointId', 'diagnosticIndex', 'manifestId', 'runId', 'outputSha256'];
    const allowed = name === 'survey_read_context' ? selectionFields : name === 'report_export' ? ['projectId', 'expectedRevision', 'adjustmentIds', 'idempotencyKey'] : name === 'survey_adjustment_read' ? ['networkId', 'adjustmentId'] : ['networkId', 'expectedRevision', 'idempotencyKey'];
    if (Object.keys(args).some(key => !allowed.includes(key))) throw deny('tool_argument_not_allowed');
    scope(args);
    if (name === 'survey_read_context') {
      for (const [key, value] of Object.entries(args)) {
        if (key === 'networkRevision' || key === 'diagnosticIndex') {
          if (!Number.isInteger(value) || value < (key === 'networkRevision' ? 1 : 0)) throw deny('invalid_context_selector');
        } else if (typeof value !== 'string' || !value.length || value.length > (key.endsWith('Sha256') ? 100 : 200)) throw deny('invalid_context_selector');
      }
      if (args.networkId !== undefined && args.networkId !== networkId) throw deny('cross_project_or_network');
      if (args.sourceSha256 !== undefined && args.sourceSha256 !== SYNTHETIC_SOURCE_SHA256) throw deny('synthetic_fixture_required');
    } else if (name === 'report_export') {
      if (args.projectId !== projectId || !Array.isArray(args.adjustmentIds) || !args.adjustmentIds.length || args.adjustmentIds.some(id => !adjustments.has(id))) throw deny('report_input_not_proven');
    } else if (args.networkId !== networkId) throw deny('cross_project_or_network');
    if (!['survey_adjustment_read', 'survey_read_context'].includes(name) && (!Number.isInteger(args.expectedRevision) || args.expectedRevision < 1)) throw deny('revision_required');
    if (args.adjustmentId && !adjustments.has(args.adjustmentId)) throw deny('adjustment_not_proven');
    if (args.idempotencyKey !== undefined && (typeof args.idempotencyKey !== 'string' || args.idempotencyKey.length > 200)) throw deny('invalid_idempotency');
    return { name, args };
  }
  async function validateRequest(body, model) {
    const json = parse(body.toString('utf8'));
    if (json?.model !== model || !Array.isArray(json.messages) || !json.messages.length || json.messages.length > 60 || json.stream !== undefined && typeof json.stream !== 'boolean') throw deny('invalid_model_or_messages');
    const topAllowed = new Set(['model', 'messages', 'tools', 'tool_choice', 'stream', 'stream_options', 'temperature', 'top_p', 'max_tokens', 'max_completion_tokens', 'thinking', 'reasoning_effort', 'frequency_penalty', 'presence_penalty', 'stop', 'parallel_tool_calls']);
    if (Object.keys(json).some(key => !topAllowed.has(key))) throw deny('request_field_not_allowed');
    if (json.tools !== undefined && (!Array.isArray(json.tools) || json.tools.length > 12)) throw deny('invalid_tool_definitions');
    const scan = structuredClone(json);
    for (const tool of scan.tools ?? []) {
      if (tool.type !== 'function') throw deny('tool_not_allowed');
      toolName(tool.function?.name);
      // Zod emits this exact static metadata; it is never a fetched resource.
      if (tool.function?.parameters?.$schema === 'https://json-schema.org/draft/2020-12/schema') delete tool.function.parameters.$schema;
    }
    scope(JSON.stringify(scan));
    if (typeof json.tool_choice === 'object') { if (json.tool_choice?.type !== 'function') throw deny('tool_not_allowed'); toolName(json.tool_choice.function?.name); }
    const stored = await evidenceLoader();
    const results = new Map(stored.messages.filter(item => item.kind === 'tool_result').map(item => [item.callId, item]));
    const pending = new Map();
    let users = 0;
    for (const message of json.messages) {
      if (!message || !['system', 'developer', 'user', 'assistant', 'tool'].includes(message.role) || message.function_call) throw deny('invalid_message');
      if (Object.keys(message).some(key => !['role', 'content', 'reasoning_content', 'tool_calls', 'tool_call_id'].includes(key)) || message.reasoning_content !== undefined && typeof message.reasoning_content !== 'string') throw deny('message_field_not_allowed');
      if (message.role !== 'assistant' && message.tool_calls || message.role !== 'tool' && message.tool_call_id) throw deny('invalid_tool_message');
      const text = contentText(message.content, message.role === 'assistant' && Boolean(message.tool_calls));
      scope(text);
      if (message.role === 'user') {
        users++;
        if (!text.includes(MARKER)) throw deny('every_user_requires_synthetic_marker');
      }
      if (message.tool_calls) {
        if (!Array.isArray(message.tool_calls) || !message.tool_calls.length || message.tool_calls.length > 4) throw deny('invalid_tool_calls');
        for (const call of message.tool_calls) {
          const expected = issued.get(call.id);
          const checked = callArguments(call);
          if (!expected || canonical(callArguments(expected)) !== canonical(checked)) throw deny('unissued_or_modified_tool_call');
          if (pending.has(call.id)) throw deny('duplicate_tool_call');
          pending.set(call.id, checked);
        }
      }
      if (message.role === 'tool') {
        const call = pending.get(message.tool_call_id);
        const result = results.get(message.tool_call_id);
        if (!call || !result || toolName(result.toolName) !== call.name || text !== (typeof result.output === 'string' ? result.output : JSON.stringify(result.output))) throw deny('tool_result_not_candidate_evidence');
        scope(result.output);
        if (call.name === 'survey_read_context' && !result.isError) {
          const context = result.output?.context;
          if (context?.projectId !== projectId || !Array.isArray(context.surveyNetworks) || context.surveyNetworks.some(network => network.id !== networkId)
            || ['datasets', 'analyses', 'citations', 'watchDrafts'].some(key => !Array.isArray(context[key]) || context[key].length)
            || !Array.isArray(context.surveyAdjustments) || context.surveyAdjustments.some(run => run.networkId !== networkId)
            || result.output.selectedNetwork && result.output.selectedNetwork.id !== networkId
            || !Array.isArray(result.output.adjustments) || result.output.adjustments.some(run => run.networkId !== networkId)) throw deny('context_not_synthetic_project');
        }
        if (result.output?.run?.projectId === projectId && result.output.run.networkId === networkId && typeof result.output.run.id === 'string') adjustments.add(result.output.run.id);
        pending.delete(message.tool_call_id);
      }
    }
    if (!users || pending.size) throw deny('incomplete_synthetic_tool_round');
    return json;
  }
  function validateResponse(bytes, contentType) {
    const calls = [];
    if (/^text\/event-stream/i.test(contentType)) {
      const byIndex = new Map();
      let done = false;
      for (const frame of bytes.toString('utf8').split(/\r?\n\r?\n/)) {
        const data = frame.split(/\r?\n/).filter(line => line.startsWith('data:')).map(line => line.slice(5).trimStart()).join('\n');
        if (!data) continue;
        if (data === '[DONE]') { done = true; continue; }
        const event = parse(data);
        if (event.error) throw deny('upstream_error_event');
        for (const choice of event.choices ?? []) {
          if (choice.index !== undefined && choice.index !== 0 || choice.delta?.function_call) throw deny('unsupported_response_choice');
          for (const delta of choice.delta?.tool_calls ?? []) {
            if (!Number.isInteger(delta.index) || delta.index < 0 || delta.index > 3) throw deny('invalid_tool_delta');
            const call = byIndex.get(delta.index) ?? { id: '', type: 'function', function: { name: '', arguments: '' } };
            if (delta.id) call.id += delta.id;
            if (delta.type && delta.type !== 'function') throw deny('tool_not_allowed');
            if (delta.function?.name) call.function.name += delta.function.name;
            if (delta.function?.arguments) call.function.arguments += delta.function.arguments;
            byIndex.set(delta.index, call);
          }
        }
      }
      if (!done) throw deny('incomplete_upstream_stream');
      calls.push(...byIndex.values());
    } else {
      const response = parse(bytes.toString('utf8'));
      if (response.error || !Array.isArray(response.choices) || response.choices.length !== 1 || response.choices[0]?.message?.function_call) throw deny('invalid_upstream_response');
      calls.push(...response.choices[0].message.tool_calls ?? []);
    }
    for (const call of calls) {
      callArguments(call);
      if (issued.has(call.id) && canonical(issued.get(call.id)) !== canonical(call)) throw deny('duplicate_upstream_call_id');
    }
    for (const call of calls) issued.set(call.id, call);
  }
  return { validateRequest, validateResponse };
}
